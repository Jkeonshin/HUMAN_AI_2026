# 1. 간단한 머신러닝 모델을 학습하고 결과(model.pkl)를 저장하는 과정

vim make_model.py    # 코드 작성
:set paste > i(입력모드 변환) > ctrl + v > esc(입력모드 종료) > :wq!(저장) > enter

----------------------------------------------------------------make_model.py
#make_model.py (학습 모델 생성)
from sklearn.ensemble import RandomForestClassifier
import numpy as np
import joblib

X = np.random.randn(200, 3)
y = (X[:,0] + X[:,1] > 0).astype(int)

model = RandomForestClassifier()
model.fit(X, y)

joblib.dump(model, "model.pkl")
print("model.pkl 생성 완료!")
----------------------------------------------------------------

# 코드 확인
cat make_model.py    # 코드 내용 보이면 정상

# 실행
python make_model.py

----------------------------------------------------------------app.py
## app.py
from fastapi import FastAPI
import joblib

app = FastAPI()
model = joblib.load("model.pkl")

@app.get("/predict")
def predict(f1: float, f2: float, f3: float):
    pred = model.predict([[f1, f2, f3]])
    return {"prediction": int(pred[0])}
----------------------------------------------------------------

# app.py 실행
uvicorn app:app --host 0.0.0.0 --port 8000

# 브라우저 테스트
http://localhost:8000/predict?f1=0.1&f2=-1&f3=2.5

## app.py
from fastapi import FastAPI
import joblib

app = FastAPI()
model = joblib.load("model.pkl")

@app.get("/predict")
def predict(f1: float, f2: float, f3: float):
    pred = model.predict([[f1, f2, f3]])
    return {"prediction": int(pred[0])}
--------------------------------------------------------

# 이미지 빌드 
> sudo docker build -t ai/model:v1 .

# 도커 서버 실행
> sudo docker run -d -p 8000:8000 ai/model:v1

#실행확인 
sudo docker ps

#결과 테스트 (web)

from sklearn.linear_model import LogisticRegression
import numpy as np
import joblib

X = np.random.randn(200, 3)
y = (X[:,0] + X[:,1] > 0).astype(int)
