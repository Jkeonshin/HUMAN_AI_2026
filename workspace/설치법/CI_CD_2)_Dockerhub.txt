1-2. train.py 생성
---------------------------------------------------------------------
import numpy as np
import pickle
from sklearn.ensemble import RandomForestClassifier

X = np.random.randn(200, 1)
y = (X[:, 0] > 0).astype(int)

model = RandomForestClassifier()
model.fit(X, y)

with open("model.pkl", "wb") as f:
    pickle.dump(model, f)

np.save("reference.npy", X)

print("model.pkl, reference.npy 생성 완료")
---------------------------------------------------------------------

실행:

python train.py


1-3. app.py 생성
---------------------------------------------------------------------
from fastapi import FastAPI
import pickle
import numpy as np

app = FastAPI()

with open("model.pkl", "rb") as f:
    model = pickle.load(f)

@app.get("/")
def home():
    return {"message": "AI Model API is running"}

@app.get("/predict")
def predict(value: float):
    X = np.array([[value]])
    pred = model.predict(X)[0]

    return {
        "input": value,
        "prediction": int(pred)
    }

FROM python:3.10-slim

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

EXPOSE 8000

CMD ["uvicorn", "app:app", "--host", "0.0.0.0", "--port", "8000"]

1-6. Docker 이미지 생성

# Docker 이미지 빌드
$> docker build -t ai-model-api:latest .

# 내 Docker Hub 계정의 저장소 이름으로 태그 추가
$> docker tag ai-model-api:latest shinjkeon/shinjkeon_docker:latest

# 이미지 확인
$> docker images

1-8. docker-compose.yml 생성
---------------------------------------------------------------------
services:
  ai-model-api:
    image: shinjkeon/shinjkeon_docker:latest
    container_name: ai-model-api
    ports:
      - "8000:8000"
    restart: always
---------------------------------------------------------------------


1-9. Docker Compose 실행

# 기존 컨테이너가 있으면 강제로 삭제
$> docker rm -f ai-model-api || true

# Docker Compose로 컨테이너 실행
$> docker compose up -d

# 실행 상태 확인
$> docker ps


1-10. 모델 API 동작 결과 확인

# 기본 API 확인
$> curl http://localhost:8000/

# 예상 결과
{"message":"AI Model API is running"}

# 모델 예측 API 확인
$> curl "http://localhost:8000/predict?value=1.5"

# 예상 결과
{"input":1.5,"prediction":1}

# Swagger API 문서
http://localhost:8000/docs


1-11. 컨테이너 중지 및 삭제

# 컨테이너 중지
$> docker stop ai-model-api

# 중지된 컨테이너 삭제
$> docker rm ai-model-api

# 실행 여부와 관계없이 컨테이너 강제 삭제
$> docker rm -f ai-model-api || true

# Docker Compose 서비스와 네트워크 종료 및 삭제
$> docker compose down


1-12. Docker 이미지 삭제

# 로컬 실행용 이미지 삭제
$> docker rmi ai-model-api:latest

# Docker Hub용 태그 이미지 삭제
$> docker rmi shinjkeon/shinjkeon_docker:latest

# 이미지 ID로 삭제하는 경우
